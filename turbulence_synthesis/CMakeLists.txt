cmake_minimum_required(VERSION 3.25)

# Turbulence synthesis (W5): procedural sub-grid swirls per camera on top of a coarse world field,
#   u_detail(x) = sum_i curl( A(x) * psi_i(x) ),  i = 1 .. layers(x)   (see src/detail/turbulence.hpp)
# CPU reference (ff_detail) and GPU version (ff_detail_gpu, shaders/turbulence.wgsl).
#
# Integration module, like froxel_grid/: the parent project provides
#   ff_gpu         shared WebGPU infrastructure (gpu/context.hpp, buffer.hpp, timer.hpp, handle.hpp)
#   glm::glm       GLM 1.0.3
#   ff_camera,     the froxel grid module (froxel_grid/, add_subdirectory before this one)
#   ff_froxel_gpu
#   doctest::doctest  optional: builds turbulence_tests and turbulence_gpu_tests
# and then calls add_subdirectory(turbulence_synthesis). Targets created here:
#   ff_froxel_link, ff_froxel_link_gpu   the only coupling to the froxel grid (src/froxel_link/)
#   ff_detail, ff_detail_gpu             the turbulence synthesis
#   turbulence_demo                      CPU demo (option FF_TURBULENCE_BUILD_APPS)
# The world field (src/detail/world_field.hpp) is a stand-in for the world solver; obstacles come in
# through the interface src/detail/obstacles.hpp (implemented by the host project).
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(TurbulenceSynthesis LANGUAGES CXX)
endif()

option(FF_TURBULENCE_BUILD_APPS "Build the turbulence synthesis demo" ON)

foreach(required IN ITEMS ff_gpu glm::glm ff_camera ff_froxel_gpu)
    if(NOT TARGET ${required})
        message(FATAL_ERROR
            "Turbulence synthesis needs the parent project's ${required} target. Create ff_gpu and glm::glm, "
            "add_subdirectory(froxel_grid), then add_subdirectory(turbulence_synthesis).")
    endif()
endforeach()
foreach(existing IN ITEMS ff_froxel_link ff_froxel_link_gpu ff_detail ff_detail_gpu)
    if(TARGET ${existing})
        message(FATAL_ERROR "${existing} already exists. This module provides it; remove the old definition.")
    endif()
endforeach()

function(ff_turbulence_warnings target)
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4 /permissive- /utf-8)
    else()
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
    endif()
    set_target_properties(${target} PROPERTIES CXX_EXTENSIONS OFF)
    target_compile_features(${target} PUBLIC cxx_std_20)
endfunction()

# Shader: own generated header, directory and namespace (does not collide with the parent's shaders.hpp).
set(generated_dir "${CMAKE_CURRENT_BINARY_DIR}/generated")
set(shader_path "${CMAKE_CURRENT_SOURCE_DIR}/shaders/turbulence.wgsl")
file(READ "${shader_path}" shader_source)
set(shader_header "// Generated by turbulence_synthesis/CMakeLists.txt. Do not edit.\n#pragma once\n#include <string_view>\nnamespace ff::turbulence_shaders {\n")
string(APPEND shader_header "inline constexpr std::string_view turbulence_wgsl = R\"WGSL(${shader_source})WGSL\";\n}\n")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${shader_path}")
file(MAKE_DIRECTORY "${generated_dir}")
file(CONFIGURE OUTPUT "${generated_dir}/ff_turbulence_shaders.hpp" CONTENT "${shader_header}" @ONLY)

# --- interface to the froxel grid (CPU / GPU) ---
add_library(ff_froxel_link STATIC src/froxel_link/froxel_link.cpp)
target_include_directories(ff_froxel_link PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/src")
target_link_libraries(ff_froxel_link PUBLIC ff_camera)
ff_turbulence_warnings(ff_froxel_link)

add_library(ff_froxel_link_gpu STATIC src/froxel_link/gpu_froxel_link.cpp)
target_link_libraries(ff_froxel_link_gpu PUBLIC ff_froxel_link ff_froxel_gpu)
ff_turbulence_warnings(ff_froxel_link_gpu)

# --- turbulence synthesis, CPU reference (no WebGPU dependency) ---
add_library(ff_detail STATIC
    src/detail/noise.cpp
    src/detail/world_field.cpp
    src/detail/turbulence.cpp
    src/detail/froxel_field.cpp
    src/detail/preview.cpp
)
target_link_libraries(ff_detail PUBLIC ff_froxel_link)
ff_turbulence_warnings(ff_detail)

# --- turbulence synthesis on the GPU ---
add_library(ff_detail_gpu STATIC src/detail/gpu_turbulence.cpp)
target_include_directories(ff_detail_gpu PRIVATE "${generated_dir}")
target_link_libraries(ff_detail_gpu PUBLIC ff_detail ff_gpu ff_froxel_link_gpu)
ff_turbulence_warnings(ff_detail_gpu)

if(FF_TURBULENCE_BUILD_APPS)
    add_executable(turbulence_demo apps/turbulence_demo/main.cpp)
    target_link_libraries(turbulence_demo PRIVATE ff_detail)
    ff_turbulence_warnings(turbulence_demo)
endif()

if(TARGET doctest::doctest)
    add_executable(turbulence_tests tests/turbulence_tests.cpp)
    target_link_libraries(turbulence_tests PRIVATE ff_detail doctest::doctest)
    ff_turbulence_warnings(turbulence_tests)
    add_test(NAME turbulence_tests COMMAND turbulence_tests)

    add_executable(turbulence_gpu_tests tests/turbulence_gpu_tests.cpp)
    target_link_libraries(turbulence_gpu_tests PRIVATE ff_detail_gpu doctest::doctest)
    ff_turbulence_warnings(turbulence_gpu_tests)
    if(COMMAND target_copy_webgpu_binaries)
        target_copy_webgpu_binaries(turbulence_gpu_tests)
    endif()
    add_test(NAME turbulence_gpu_tests COMMAND turbulence_gpu_tests)
endif()
